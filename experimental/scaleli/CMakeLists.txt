cmake_minimum_required(VERSION 3.16)
project(scaleli VERSION 0.1.0 LANGUAGES CXX)
option(SCALELI_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(SCALELI_NATIVE "Enable host-specific CPU tuning (record this in experiments)" OFF)
option(SCALELI_EXTERNAL "Build optional official ALEX / dynamic PGM adapters" OFF)
set(SCALELI_EXTERNAL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external" CACHE PATH "Fetched upstream repositories")
add_library(scaleli INTERFACE)
target_include_directories(scaleli INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(scaleli INTERFACE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(scaleli INTERFACE -Wall -Wextra -Wconversion -Wno-sign-conversion -Wno-misleading-indentation)
 if(SCALELI_NATIVE)
  target_compile_options(scaleli INTERFACE -march=native)
 endif()
 if(SCALELI_SANITIZE)
  target_compile_options(scaleli INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(scaleli INTERFACE -fsanitize=address,undefined)
 endif()
endif()
enable_testing()
add_executable(scaleli_tests tests/test_main.cpp)
target_link_libraries(scaleli_tests PRIVATE scaleli)
add_test(NAME scaleli_core COMMAND scaleli_tests)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/benchmark.cpp")
 add_executable(scaleli_bench src/benchmark.cpp)
 target_link_libraries(scaleli_bench PRIVATE scaleli)
 if(SCALELI_EXTERNAL)
  target_compile_definitions(scaleli_bench PRIVATE SCALELI_EXTERNAL=1)
  if(NOT EXISTS "${SCALELI_EXTERNAL_DIR}/ALEX/src/core/alex.h" OR NOT EXISTS "${SCALELI_EXTERNAL_DIR}/PGM-index/include/pgm/pgm_index_dynamic.hpp")
   message(FATAL_ERROR "Run tools/fetch_external.py first; external sources are intentionally not bundled.")
  endif()
  add_library(scaleli_external STATIC integrations/external/bridge.cpp)
  set_target_properties(scaleli_external PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
  target_include_directories(scaleli_external PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include ${SCALELI_EXTERNAL_DIR}/ALEX/src/core ${SCALELI_EXTERNAL_DIR}/PGM-index/include)
  if(SCALELI_SANITIZE)
   target_compile_options(scaleli_external PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
  target_link_libraries(scaleli_bench PRIVATE scaleli_external)
 endif()
endif()
# Dataset hardness metrics (AIDB 2026 protocol; LIPP FMCD and PGM PLA ports) and their tests.
find_package(Threads)
add_executable(scaleli_hardness src/hardness.cpp)
target_link_libraries(scaleli_hardness PRIVATE scaleli)
add_executable(scaleli_hardness_tests tests/test_hardness.cpp)
target_link_libraries(scaleli_hardness_tests PRIVATE scaleli)
if(Threads_FOUND)
 target_link_libraries(scaleli_hardness PRIVATE Threads::Threads)
 target_link_libraries(scaleli_hardness_tests PRIVATE Threads::Threads)
endif()
set(SCALELI_HARDNESS_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/../../evidence/dense_sparse_fixture_uint64")
if(NOT EXISTS "${SCALELI_HARDNESS_FIXTURE}")
 set(SCALELI_HARDNESS_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/data/dense_sparse_fixture_uint64")
endif()
add_test(NAME scaleli_hardness_tests COMMAND scaleli_hardness_tests $<TARGET_FILE:scaleli_hardness> "${SCALELI_HARDNESS_FIXTURE}" "${CMAKE_CURRENT_SOURCE_DIR}/results/flows/dense_sparse_2D2H2L.txt")
add_executable(scaleli_example examples/basic.cpp)
target_link_libraries(scaleli_example PRIVATE scaleli)
add_test(NAME scaleli_example COMMAND scaleli_example)
