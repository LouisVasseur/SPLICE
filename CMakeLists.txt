cmake_minimum_required(VERSION 3.16)
project(scaleli_sota_walkthrough LANGUAGES CXX)
enable_testing()
add_executable(radix_walkthrough native/radix_walkthrough.cpp)
target_compile_features(radix_walkthrough PRIVATE cxx_std_17)
target_include_directories(radix_walkthrough PRIVATE third_party/radix_spline/include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(radix_walkthrough PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
option(WALKTHROUGH_SANITIZE "Enable address/undefined sanitizers for the native walkthrough" OFF)
if(WALKTHROUGH_SANITIZE)
 target_compile_options(radix_walkthrough PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(radix_walkthrough PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME radix_differential COMMAND radix_walkthrough --self-test)
add_subdirectory(experimental/scaleli)
